<?php
/**

 *      //简单使用PDO查询，并返回所有结果
 *      $a = pdos::select('test_blog')->where(array('user_id'=>2))->exec()->getAll();
 *      
 *      //删除,并返回是否执行成功
 *      $b = pdos::delete('test_blog')->where(array('user_id'=>1))->exec()->isTrue();
 *      
 *      //插入
 *      $pdo = pdos::insert('test_blog',array('user_id'=>1,'blog'=>'内容2'))->exec();      
 *      $c = $pdo->insertId();//获取插入ID

 *      //更新,不返回有效使用

 *      GROUP BY u.user_id ORDER BY b.blog_id DESC LIMIT ?,?';
 *      $k = pdos::query($sql, array(1, 2, 3, 158, 'a', 0, 5))->getAll();
 *      
 *      //事务：
 *      $pdo = pdos::insert('test_blog',array('user_id'=>2,'blog'=>'内容2'));
 *      $pdo->begin();
 *      $m = $pdo->exec()->isTrue();
 *      $m ? $pdo->commit() : $pdo->rollback();
 *      
 *      var_dump($a,$b,$c,$d,$e,$f,$g,$h,$i,$j,$k,$m);

class pdos
{
    /**
     * 增删改查静态入口
     * 
     * @version 1.0    2016-06-28
     * @author chendong@yolo24.com
     */    
    static public function __callStatic($func, $param)
    {
        $database = array('select'=>1, 'insert'=>0, 'delete'=>0, 'update'=>0, 'query'=>0);
        if(isset($database[$func]))
        {
            $pdo = new PDO_LIB($database[$func]); 
            return call_user_func_array(array(&$pdo, $func), $param);
        }
        else
        {
            return false;
        }
    }
}

/**
 * PDO_LIB PDO封装库
 * 
 * 1,需要database.php有统一配置
 * 2,支持多服务器
 * 3,支持读写分离
 * 
 *一般使用配置database.php：
 *   array(
 *       'db_type' => 'mysql',       // 数据库类型
 *       'db_host' => '10.126.53.37',// 数据库服务器地址
 *       'db_name' => 'gome_wap',    // 数据库名
 *       'db_user' => 'root',        // 数据库用户名
 *       'db_pwd' => 'root',         // 数据库密码
 *       'db_port' => '7308',        // 数据库端口
 *       'db_prefix' => 'wap_',      // 数据库表前缀 //因表名配置不统一,暂时不起作用
 *   );
 *   
 * 多台数据库服务器使用配置：
 * array(
 *     $database1,
 *     $database2,
 *     $database3,
 *     ......
 * )
 *
 * 主从使用配置:
 * array(
 *     //主服务器组
 *     ayyay(
 *         $database1,
 *         ......
 *     ),
 *     //从服务器组
 *     array(
 *         $database3,
 *         ......
 *     )
 * )
 *
 * 4,资源单例连接
 * 5,支持链式操作
 * 6,支持事务
 * 7,除了query操作原生SQL，全部使用预处理
 * 8,除了增删改查，支持联表查询
 * 9,支持查询中使用> < != = >= <= IN LIKE
 * 10,支持group
 * 11,有限度的支持OR
 * @version 1.0
 * @author  <cd>chendong@yolo24.com
 * @createtime 2016-06-27
 */
class PDO_LIB
{
    static private  $_pdos;
    static private  $_pre = '';

    private $_pdo;
    private $_stmt;
    private $_config;
    private $_open_rw;

    private $_sql  = '';
    private $_data = array();
    private $_isTrue = false;
    private $_result = false;

    //实例化PDO
    public function __construct($is_main = 0)
    {
        $this->config();

        $is_main = $this->_open_rw ? $is_main : 0;

        if (!isset(self::$_pdos[$is_main]))
        {
            $num    = count($this->_config[$is_main]);
            $index  = mt_rand(0, $num-1);
            $config = $this->_config[$is_main][$index];

            $dsn    = $config['db_type'] .':host='.$config['db_host'].';port='.$config['db_port'].';dbname='.$config['db_name'].';charset=utf8';
            $opts = array(
                        //PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,//设置抛出错误
                        PDO::ATTR_AUTOCOMMIT => 1,                  //开启自动提交
                        PDO::ATTR_TIMEOUT => 60,                    //设置超时时间
                        PDO::ATTR_PERSISTENT => false,              //持久连接;虽然非常高效,但容易引起问题：表锁、事物阻塞、闲置连接无限循环,故不开启
                        PDO::MYSQL_ATTR_USE_BUFFERED_QUERY => true, //开启mysql缓冲查询，使用多重查询结果很重要
                        PDO::ATTR_EMULATE_PREPARES => false,        //不开启本地预处理，交给mysql转义，防止5.4以下部分版本bug
                        PDO::MYSQL_ATTR_INIT_COMMAND => 'set names utf8' ,  //设置提交及结果字符集编码，同时防止部分php版本对dsn设置不完全生效
                        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,   //设置fetchAll结果集为关联索引数组
            );

            self::$_pdos[$is_main] = new PDO_GOME($dsn, $config['db_user'], $config['db_pwd'], $opts);

            //前缀不统一,暂不启用
            //if (!empty($config['db_prefix']))
            //{
            //    self::$_pre = $config['db_prefix'];
            //}
        }

        $this->_pdo = & self::$_pdos[$is_main];
    }

    /**
     * 组合数据库配置
     */    
    private function config()
    {
        $this->_open_rw = false;

        if (isset($config[0]))
        {
            if (isset($config[0][0]))
            {
               $this->_config = $config;

               $this->_open_rw = true; //开启主从,读写分离
            }
            else
            {
                $this->_config = array($config,$config);
            }
        }
        else
        {
            $config = Config::get('database');
            $this->_config = array(array($config), array($config));
        }
    }

    public function begin()
    {
        return $this->_pdo->beginTransaction();
    }

    public function commit()
    {
        return $this->_pdo->commit();
    }

    public function rollback()
    {
        return $this->_pdo->rollback();
    }

    /**
     * 增加一条记录
     * str $table 表名
     * arr $param array('字段名1'=>'字段值1', '字段名2'=>'字段值2');
     * @version 1.0    2016-06-24
     * @author chendong@yolo24.com
     */
    public function insert($table = '', $param = array())
    {
        $this->_sql .= 'INSERT INTO '. self::$_pre . $table .' (';

        $values = 'VALUES (';

        foreach ($param as $key => $val) 
        {
            $this->_sql .= '`'.$key . '`, ';
            $values .= '?, ';
            $this->_data[]    =  $val;
        }

        $this->_sql = rtrim($this->_sql, ', ') . ') ' . rtrim($values, ', ') . ');';

        return $this;
    }

    /**
     * 删除记录
     * $table 表名
     * @version 1.0    2016-06-24
     * @author chendong@yolo24.com
     */
    public function delete($table = '')
    {
        $this->_sql .= 'DELETE FROM '. self::$_pre . $table . ' ';

        return $this;
    }

    /**
     * 更新记录
     * str $table 表名
     * arr $param array('字段名1'=>'字段值1', '字段名2'=>'字段值2');
     * @version 1.0    2016-06-24
     * @author chendong@yolo24.com
     */
    public function update($table = '', $param = array())
    {
        $this->_sql .= 'UPDATE '. self::$_pre . $table .' SET ';

        foreach ($param as $key => $val)
        {
            $this->_sql   .= '`' . $key . '` = ?, ';
            $this->_data[]    =  $val;
        }

       $this->_sql = rtrim($this->_sql, ', ') . ' ';

        return $this;
    }

    /**
     * 查询
     * str      $table 表名
     * mixed    $param array('字段名1', '字段名2') | '字段名,字段名' | 不传默认*;
     * @version 1.0    2016-06-24
     * @author chendong@yolo24.com
     */
    public function select($table = '', $param = array())
    {
        $this->_sql .= 'SELECT ';

        if (is_string($param))
        {
            $this->_sql .= $param .' ';
        }
        elseif (empty($param))
        {
            $this->_sql .= '* ';
        }
        else
        {
            $this->_sql .= implode(',', $param) . ' ';
        }

        $this->_sql .= 'FROM '. self::$_pre . $table .' ';

        return $this;
    }


    /**
     * 使用自定义预处理SQL
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function query($sql, $param)
    {
        $this->_sql     = $sql;
        $this->_data    = $param;
        return $this->exec();       
    }

    /**
     * order by
     * str $by '字段名 DESC' | '字段名 ASC'
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function where($param = array())
    {
        $this->_sql .= 'WHERE ';
        if($param)
        {
            foreach ($param as $key => $val) 
            {
                $this->whereStr($key, $val);
                $this->_sql .= 'AND ';
            }
            $this->_sql = rtrim($this->_sql, 'AND ') .' ';            
        }
        else
        {
             $this->_sql .= '1 ';
        }

        return $this;
    }

    /**
     * order by
     * str $by '字段名 DESC' | '字段名 ASC'
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function andOr($param = array())
    {
        if($param)
        {
            $this->_sql .= 'AND ( ';
            foreach ($param as $key => $val) 
            {
                $this->whereStr($key, $val);
                $this->_sql .= 'OR ';
            }

            $this->_sql = rtrim($this->_sql, 'OR ') .' ) ';
        }
                return $this;
    }

     /**
     * 支持 != = > < IN LIKE >= <=
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */       
    private function whereStr($key, $val)
    {
        $arr    = array('>', '<', '!', '=');
        $first  = strtoupper(substr($val,0,1));

        if(in_array($first, $arr) || strtoupper(substr($val,0,3)) == 'IN ' || strtoupper(substr($val,0,5)) == 'LIKE ')
        {
            $index = strpos($val, ' ');
            $this->_sql .= $key .' '.substr($val, 0, $index).' ';

            $value = substr($val, ++$index);

            if($first == 'I')
            {
                $this->_sql .= '(';
                $arr = explode(',', $value);
                foreach ($arr as $k => $v) {
                    $this->_sql .= '?, ';
                    $this->_data[]  =  $v;
                }
                $this->_sql = rtrim($this->_sql, ', ') .') ';
            }
            else
            {
                $this->_sql .= '? '; 
                $this->_data[]  =  $value;
            }
        }
        else
        {
            $this->_sql .= $key .' = ? ';
            $this->_data[]  =  $val;
        }
    }

    /**
     * group by
     * str $by '字段名'
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function group($by)
    {
        $this->_sql .= 'GROUP BY '. $by .' ';

        return $this;
    }

    /**
     * order by
     * str $by '字段名 DESC' | '字段名 ASC'
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function order($by)
    {
        $this->_sql .= 'ORDER BY '. $by .' ';

        return $this;
    }

    /**
     * limit
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function limit($start, $num)
    {
        $this->_sql .= 'LIMIT ?,? ';
        $this->_data[]  =  $start;
        $this->_data[]  =  $num;
        return $this;
    }

    /**
     * join
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function join($table='', $param = array(), $place = 'LEFT')
    {
        $this->_sql .= $place .' JOIN '. self::$_pre . $table .' ON ';

        if($param)
        {
            foreach($param as $key=>$val)
            {
                $this->whereStr($key, $val);
                $this->_sql .= 'AND '; 
            }
            $this->_sql = rtrim($this->_sql, 'AND ') .' ';
        }

        return $this;
    }

    /**
     * join
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function on($param = array())
    {
        if($param)
        {
            foreach ($param as $key => $val) 
            {
                $this->_sql .= $key .' = '. $val .' AND ';
            }

            $this->_sql = rtrim($this->_sql, 'AND ') .' ';
        }

        return $this;
    }

    /**
     * 执行sql
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function exec()
    {
        try
        {
            $this->_stmt   = $this->_pdo->prepare($this->_sql);

            $this->_isTrue = $this->_stmt ? $this->_stmt->execute($this->_data) : false;
        }
        catch (PDOException $e) 
        {
            $this->log('query error : ' . $e->getMessage());

            $this->_isTrue = false;
        }

        return $this;
    }

    /**
     * 输出上一次执行是否成功
     * @version 1.0    2016-06-28
     * @author chendong@yolo24.com
     */    
    public function isTrue()
    {
        return $this->_isTrue;
    }      

    /**
     * 返回上一条insert的SQL的自增id
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function insertId()
    {
        return $this->_isTrue ? intval($this->_pdo->lastInsertId()) : 0;
    }

    /**
     * 返回上一条SQL的影响行数
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function updateNum()
    {
        return $this->_isTrue ? intval($this->_stmt->rowCount()) : 0;
    }

    /**
     * 结果集总数
     * @version 1.0    2016-06-24
     * @author chendong@yolo24.com
     */
    public function selectNum()
    {
        return $this->_isTrue ? $this->_pdo->query("SELECT FOUND_ROWS()")->fetchColumn() : 0;
    }

    /**
     * 返回select的一行结果集
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function getRow()
    {
        $result = $this->getAll();
        $result = $result ? $result[0] : false;
        return $result;
    }

    /**
     * 返回select的所有结果集
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    public function getAll()
    {
        if (empty($this->_result) && $this->_isTrue && strtoupper(substr($this->_sql, 0, 6) == 'SELECT'))
        {
            $this->_result = $this->_stmt->fetchAll();
        }
        return $this->_result;
    }

    /**
     * 按sql获取第一条数据的第一列到值
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */ 
    public function getOne() 
    {
        $result = $this->getRow();
        $result = $result ? reset($result) : false;
        return $result;
    }

    /**
     * 输出SQL 
     * @version 1.0    2016-06-28
     * @author chendong@yolo24.com
     */    
    public function getSql()
    {
        return $this->_sql;
    }

    /**
     * 输出预处理的数组
     * @version 1.0    2016-06-28
     * @author chendong@yolo24.com
     */    
    public function getArr()
    {
        return $this->_data;
    }  

    /**
     * 记录日志
     * @version 1.0    2016-06-27
     * @author chendong@yolo24.com
     */
    function log($str)
    {
        file_put_contents(LOG_PATH . '/pdo_error.log', date('Y-m-d H:i:s').'   '.$str . "\n", FILE_APPEND);
    }
}

/**
 * PDO_GOME PDO类重载优化
 * 1,优化两次开启事务
 * 2,防止暴露连接信息
 * 
 * @version 1.0
 * @author  <cd>chendong@yolo24.com
 * @createtime 2016-06-24
 */
class PDO_GOME extends PDO
{
    private $transactionCounter = 0; //防止两次开启事务报错

    public function __construct($dsn, $user='', $pwd='', $opts=array())
    {
        //暂时改变php的异常处理，防止暴露连接信息
        set_exception_handler(array(__CLASS__, 'exception_handler'));

        parent::__construct($dsn, $user, $pwd, $opts);

        restore_exception_handler();
    }

    public static function exception_handler($e)
    {
        $error = $e->getMessage();
        file_put_contents(LOG_PATH . '/pdo_error.log', date('Y-m-d H:i:s').'   '.$error . "\n", FILE_APPEND);
        exit('ERROR:'.$error);
    }

    function beginTransaction()
    {
        if (!$this->transactionCounter++)
        {
            return parent::beginTransaction();
        }
        return $this->transactionCounter >= 0;
    }

    function commit()
    {
        if (!--$this->transactionCounter)
        {
            return parent::commit();
        }
        return $this->transactionCounter >= 0;
    }

    function rollback()
    {
        if ($this->transactionCounter >= 0)
        {
            $this->transactionCounter = 0;
            return parent::rollback();
        }
        $this->transactionCounter = 0;
        return false;
    }
}
